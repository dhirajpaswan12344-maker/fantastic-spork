    <script>
        const canvas = document.getElementById("gameCanvas");
        const ctx = canvas.getContext("2d");

        let player = { x: 160, y: 430, width:    40, height: 40, color: "#00ffcc" };
        let bullets = [];
        let enemies = [];
        let score = 0;

        // Spawn enemies
        setInterval(() => {
            let ex = Math.random() * (canvas.width - 30);
            enemies.push({ x: ex, y: 0, width: 30, height: 30, speed: 2 });
        }, 1500);

        // Shoot on click/touch
        window.addEventListener("click", () => {
            bullets.push({ x: player.x + player.width / 2 - 4, y: player.y, width: 8, height: 15, speed: 6 });
        });

        // Move player with touch
        window.addEventListener("touchmove", (e) => {
            let rect = canvas.getBoundingClientRect();
            let touchX = e.touches[0].clientX - rect.left;
            if (touchX > 0 && touchX < canvas.width - player.width) {
                player.x = touchX;
            }
        });

        function update() {
            // Move bullets
            for (let i = bullets.length - 1; i >= 0; i--) {
                bullets[i].y -= bullets[i].speed;
                if (bullets[i].y < 0) bullets.splice(i, 1);
            }

            // Move enemies & check collision
            for (let i = enemies.length - 1; i >= 0; i--) {
                enemies[i].y += enemies[i].speed;

                // Bullet hit enemy
                for (let j = bullets.length - 1; j >= 0; j--) {
                    if (
                        bullets[j] &&
                        enemies[i] &&
                        bullets[j].x < enemies[i].x + enemies[i].width &&
                        bullets[j].x + bullets[j].width > enemies[i].x &&
                        bullets[j].y < enemies[i].y + enemies[i].height &&
                        bullets[j].y + bullets[j].height > enemies[i].y
                    ) {
                        enemies.splice(i, 1);
                        bullets.splice(j, 1);
                        score += 10;
                        break;
                    }
                }

                // Game over condition
                if (enemies[i] && enemies[i].y > canvas.height) {
                    enemies.splice(i, 1);
                }
            }
        }

        function draw() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // Draw Player
            ctx.fillStyle = player.color;
            ctx.fillRect(player.x, player.y, player.width, player.height);

            // Draw Bullets
            ctx.fillStyle = "#ffcc00";
            bullets.forEach(b => ctx.fillRect(b.x, b.y, b.width, b.height));

            // Draw Enemies
            ctx.fillStyle = "#ff4444";
            enemies.forEach(e => ctx.fillRect(e.x, e.y, e.width, e.height));

            // Draw Score
            ctx.fillStyle = "#fff";
            ctx.font = "16px Arial";
            ctx.fillText("Score: " + score, 20, 30);
        }

        function loop() {
            update();
            draw();
            requestAnimationFrame(loop);
        }

        loop();
    </script>
